### 功能2：存储兼容性检查

**场景**：入库时自动检查

```javascript
import { checkStorageCompatibility } from '@/api/ghs'

// 在入库表单提交前调用
async function beforeStockIn(reagentId, locationId) {
  const result = await checkStorageCompatibility({
    reagentId,
    locationId,
    operatorId: this.currentUser.id,
    operatorName: this.currentUser.name
  })
  
  if (result.data.blocked) {
    this.$message.error('存储位置不兼容：' + result.data.warnings.join('; '))
    return false
  }
  
  if (result.data.warnings.length > 0) {
    const confirmed = await this.$confirm(
      '存在以下警告，是否继续？\n' + result.data.warnings.join('\n'),
      '警告',
      { type: 'warning' }
    )
    return confirmed
  }
  
  return true
}
```

### 功能6：智能预测

**场景**：查看即将需要采购的试剂

```javascript
import { getUpcomingOrders } from '@/api/forecasting'

async function checkUpcomingOrders() {
  const result = await getUpcomingOrders()
  
  if (result.data.length > 0) {
    // 显示提醒
    this.$notify({
      title: '采购提醒',
      message: `有 ${result.data.length} 种试剂即将耗尽，请及时采购`,
      type: 'warning',
      duration: 0
    })
    
    // 显示列表
    this.upcomingOrders = result.data.map(item => ({
      reagentName: item.reagentName,
      currentStock: item.currentStock,
      daysLeft: item.daysUntilDepletion,
      orderDate: item.recommendedOrderDate,
      quantity: item.recommendedOrderQuantity,
      confidence: item.confidenceLevel + '%'
    }))
  }
}
```

## 📱 移动端扫码功能

推荐使用 `vue-qrcode-reader` 组件：

```bash
npm install vue-qrcode-reader
```

```vue
<template>
  <div>
    <qrcode-stream @decode="onDecode" @init="onInit" />
    
    <div v-if="result">
      <h3>扫描结果</h3>
      <p>二维码：{{ result.qrCode }}</p>
      <p>状态：{{ result.status }}</p>
      <p>剩余量：{{ result.remaining }}</p>
    </div>
  </div>
</template>

<script>
import { QrcodeStream } from 'vue-qrcode-reader'
import { scanQrCode } from '@/api/lifecycle'

export default {
  components: { QrcodeStream },
  
  data() {
    return {
      result: null
    }
  },
  
  methods: {
    async onDecode(qrCode) {
      const result = await scanQrCode(
        qrCode,
        this.$store.state.user.id,
        this.$store.state.user.name
      )
      
      this.result = result.data
    },
    
    onInit(promise) {
      promise.catch(console.error)
    }
  }
}
</script>
```

## 🐛 常见问题

### 问题1：数据库脚本执行失败

**原因**：可能是字符集问题或现有数据冲突

**解决**：
```bash
# 备份现有数据
mysqldump -u root -p reagent_management > backup.sql

# 重新执行脚本
mysql -u root -p reagent_management < database/modules_extension.sql
```

---

### 问题2：后端编译失败

**原因**：缺少依赖或版本不兼容

**解决**：
```bash
# 清理Maven缓存
mvn clean

# 重新下载依赖
mvn dependency:purge-local-repository

# 重新编译
mvn clean install
```

---

### 问题3：前端API调用失败

**原因**：跨域问题或请求路径错误

**解决**：
1. 检查后端CORS配置（已配置在`CorsConfig.java`）
2. 检查API路径是否正确（应该是`/backend/xxx`）
3. 检查浏览器控制台错误信息
